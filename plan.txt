        checkIfAllPrepared()

        val allPrepared = currentOrder?.items?.all { item ->
            itemPreparedStates[item.id] == true
        } ?: false
        
        if (allPrepared && currentOrder != null) {
            viewModel.markPrepared(currentOrder!!.id)
        }
    }
