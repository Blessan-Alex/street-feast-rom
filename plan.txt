### DB: allow alter to keep same table
File: `supabase/migrations/0015_enhance_alter_order.sql` (or new migration)
```sql
-- allow a dine-in alter to reuse its current table without raising occupied
create or replace function alter_order(...)
returns void
language plpgsql
as $$
declare
  v_table int;
  v_store uuid;
begin
  select table_number, store_id into v_table, v_store from orders where id = p_order_id;
  if v_table is not null then
    -- only block if another order holds the table
    if exists (
      select 1 from orders
      where store_id = v_store
        and table_number = v_table
        and id <> p_order_id
        and status in ('Created','Accepted','InKitchen','Prepared')
    ) then
      raise exception 'Table % is already occupied', v_table;
    end if;
  end if;
  ...
end;
$$;
```
